using UnityEngine;
using UnityEngine.InputSystem; // Required for the New Input System

public class SnakeMovement : MonoBehaviour
{

    public float speed = 5f;

    private Vector2 moveDirection = Vector2.up;

    void Update()
    {
        HandleInput();
        Move();
    }

    private void HandleInput()
    {
        Keyboard keyboard = Keyboard.current;
        if (keyboard == null) return; // Safety check if no keyboard is connected

        // Change to Up (Only if not currently moving Down)
        if ((keyboard.upArrowKey.wasPressedThisFrame || keyboard.wKey.wasPressedThisFrame) && moveDirection != Vector2.down)
        {
            moveDirection = Vector2.up;
        }
        // Change to Down (Only if not currently moving Up)
        else if ((keyboard.downArrowKey.wasPressedThisFrame || keyboard.sKey.wasPressedThisFrame) && moveDirection != Vector2.up)
        {
            moveDirection = Vector2.down;
        }
        // Change to Left (Only if not currently moving Right)
        else if ((keyboard.leftArrowKey.wasPressedThisFrame || keyboard.aKey.wasPressedThisFrame) && moveDirection != Vector2.right)
        {
            moveDirection = Vector2.left;
        }
        // Change to Right (Only if not currently moving Left)
        else if ((keyboard.rightArrowKey.wasPressedThisFrame || keyboard.dKey.wasPressedThisFrame) && moveDirection != Vector2.left)
        {
            moveDirection = Vector2.right;
        }
    }

    private void Move()
    {
        // Continuously moves the snake in the active direction
        transform.Translate(moveDirection * speed * Time.deltaTime);
    }
}
